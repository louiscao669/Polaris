-- organization objects

CREATE TABLE IF NOT EXISTS `DESCRIBE organization;DB` (
    `org_id` int auto_increment primary key,
    `name` varchar(255) not null,
    `description` text
);

CREATE TABLE IF NOT EXISTS `users` (
    `id` int auto_increment primary key,
    `first` varchar(100) not null,
    `last` varchar(100) not null,
    `age` int,
    `email` varchar(255) not null,
    `username` varchar(100) not null unique,
    `password_hash` varchar(255) not null
);

CREATE TABLE IF NOT EXISTS `organization_role` (
    `org_id` int not null,
    `role` varchar(100) not null,
    `description` text,
    primary key (`org_id`, `role`),
    constraint `fk_org_role_org`
        foreign key (`org_id`) references `organization`(`org_id`)
        on delete cascade
);

CREATE TABLE IF NOT EXISTS `organization_token` (
    `token_id` int auto_increment primary key,
    `org_id` int not null,
    `name` varchar(255) not null,
    `description` text,
    constraint `fk_org_token_org`
        foreign key (`org_id`) references `organization`(`org_id`)
        on delete cascade
);

CREATE TABLE IF NOT EXISTS `organization_leader` (
    `org_id` int primary key,
    `user_id` int not null,
    constraint `fk_org_leader_org`
        foreign key (`org_id`) references `organization`(`org_id`)
        on delete cascade,
    constraint `fk_org_leader_user`
        foreign key (`user_id`) references `users`(`id`)
        on delete cascade
);

-- market objects

CREATE TABLE IF NOT EXISTS `market` (
    `id` int auto_increment primary key,
    `event_id` int not null,
    `question` varchar(500) not null,
    `is_open` boolean not null default true,
    `created_by` int not null,

    `close_at` timestamp null,

        on delete cascade

-- async operations / event bus support

CREATE TABLE IF NOT EXISTS `operations` (
    `operation_id` char(36) not null,
    `topic` varchar(255) not null,
    `status` enum('queued', 'processing', 'succeeded', 'failed', 'dead') not null default 'queued',
    `envelope_json` json not null,
    `result_json` json null,
    `error_message` text null,
    `kafka_partition` int null,
    `kafka_offset` bigint null,
    `created_at` timestamp not null default current_timestamp,
    `updated_at` timestamp not null default current_timestamp on update current_timestamp,
    primary key (`operation_id`),
    key `idx_operations_status_created` (`status`, `created_at`),
    key `idx_operations_topic_created` (`topic`, `created_at`)
);

CREATE TABLE IF NOT EXISTS `processed_events` (
    `event_id` char(36) not null,
    `consumer_group` varchar(255) not null,
    `topic` varchar(255) not null,
    `kafka_partition` int not null,
    `kafka_offset` bigint not null,
    `status` enum('received', 'applied', 'skipped_duplicate', 'failed') not null default 'received',
    `processed_at` timestamp null default null,
    `error_message` text null,
    primary key (`event_id`, `consumer_group`),
    unique key `uniq_consumer_partition_offset` (`consumer_group`, `topic`, `kafka_partition`, `kafka_offset`),
    key `idx_processed_events_processed_at` (`processed_at`)
);
